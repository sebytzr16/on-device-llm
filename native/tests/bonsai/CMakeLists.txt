cmake_minimum_required(VERSION 3.22)
project(bonsai_verification LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(BONSAI_PREBUILT_DIR)
  # Reuse bridge objects; compiling all attention kernels twice is expensive.
  set(prebuilt_libs ggml ggml-base ggml-cpu llama llama-common llama-common-base cpp-httplib vendor-hash)
  if(GGML_CUDA)
    list(APPEND prebuilt_libs ggml-cuda)
  endif()
  foreach(name IN LISTS prebuilt_libs)
    file(GLOB_RECURSE candidates "${BONSAI_PREBUILT_DIR}/llama.cpp/${name}.lib")
    list(LENGTH candidates count)
    if(NOT count EQUAL 1)
      message(FATAL_ERROR "Expected one ${name}.lib in BONSAI_PREBUILT_DIR")
    endif()
    add_library(${name} STATIC IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${candidates}")
  endforeach()
  set_property(TARGET ggml APPEND PROPERTY INTERFACE_LINK_LIBRARIES ggml-base ggml-cpu)
  if(GGML_CUDA)
    find_package(CUDAToolkit REQUIRED)
    set_property(TARGET ggml APPEND PROPERTY INTERFACE_LINK_LIBRARIES ggml-cuda)
    set_property(TARGET ggml-cuda APPEND PROPERTY INTERFACE_LINK_LIBRARIES CUDA::cudart CUDA::cublas CUDA::cuda_driver)
  endif()
  set_property(TARGET llama APPEND PROPERTY INTERFACE_LINK_LIBRARIES ggml)
  set_property(TARGET llama-common APPEND PROPERTY INTERFACE_LINK_LIBRARIES llama llama-common-base cpp-httplib vendor-hash ws2_32 crypt32)
  include_directories(../../llama.cpp/include ../../llama.cpp/ggml/include ../../llama.cpp/common ../../llama.cpp/vendor)
  add_executable(llama-simple ../../llama.cpp/examples/simple/simple.cpp)
  target_link_libraries(llama-simple PRIVATE llama)
  add_executable(llama-completion ../../llama.cpp/tools/completion/main.cpp ../../llama.cpp/tools/completion/completion.cpp)
  add_executable(llama-perplexity ../../llama.cpp/tools/perplexity/main.cpp ../../llama.cpp/tools/perplexity/perplexity.cpp)
  add_executable(llama-bench ../../llama.cpp/tools/llama-bench/main.cpp ../../llama.cpp/tools/llama-bench/llama-bench.cpp)
  add_executable(llama-quantize ../../llama.cpp/tools/quantize/main.cpp ../../llama.cpp/tools/quantize/quantize.cpp)
  foreach(name llama-completion llama-perplexity llama-bench llama-quantize)
    target_link_libraries(${name} PRIVATE llama-common)
  endforeach()
else()
  add_subdirectory(../../llama.cpp llama.cpp)
endif()
add_executable(bonsai-numerics numerics.cpp)
target_include_directories(bonsai-numerics PRIVATE ../../llama.cpp/ggml/src)
target_link_libraries(bonsai-numerics PRIVATE ggml ggml-cpu)
if(GGML_CUDA)
  target_compile_definitions(bonsai-numerics PRIVATE BONSAI_CUDA)
  target_link_libraries(bonsai-numerics PRIVATE ggml-cuda)
endif()
